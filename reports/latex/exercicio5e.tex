    for i in range(1, img.shape[0] - 1):
        for j in range(1, img.shape[1] - 1):
            up = (img[i, j] != img[i + 1, j]).any()
            down = (img[i, j] != img[i - 1, j]).any()
            left = (img[i, j] != img[i, j - 1]).any()
            right = (img[i, j] != img[i, j + 1]).any()
